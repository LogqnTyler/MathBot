\section{Anti-derivatives}
\begin{enumerate}
    \item Compute an anti-derivative of 
    \begin{enumerate}
        \item $\ds f(x) = e^x - x^2 + \frac{1}{x}.$
        \item $\ds g(x) = \frac{x^3+ 2x + 10x^9}{x^2}.$
    \end{enumerate}
    \item Compute 
    \begin{enumerate}
        \item $\ds \int (x^4 - x^3 + x^2)(x^2 + 1)\,dx.$
        \item $\ds \int \left(e^x  + \frac{9}{x}\right)\,dx.$
    \end{enumerate}    


\end{enumerate}

   \section{Summation}
   \begin{enumerate}

    
    \item Rewrite in summation notation -you don't need to solve!-:
    \begin{enumerate}
        \item $\ds 2 + 4 + 6 + 8 $
        \item $\ds 1+2+3+4 + \cdots + n$
        \item $\ds 2 - 4 + 6 - 8 + 10.$
    \end{enumerate}

    \item Knowing that $\ds \sum_{i=1}^{n}a_i=5,\sum_{i=1}^{n}b_i=0,\sum_{i=1}^{n}c_i=-1$, use the properties of summation to compute the values of the following summations
    \begin{enumerate}
        \item $\ds \sum_{i=1}^n \frac{a_i+b_i}{2}$.
        \item $\ds \sum_{i=1}^n 3a_i-b_i-3c_i+10$.
    \end{enumerate}
        \item (Bonus!) Compute explicitly the following sums 
    \begin{enumerate}
        \item $\ds \sum_{k = 1}^4 (-1)^k k.$
        \item $\ds \sum_{k = 0}^3 (-1)^{k+1} (k+1).$
        \item $\ds \sum_{k = 100}^{104} (-1)^{k-99} (k-99)$
    \end{enumerate}
    \end{enumerate}
