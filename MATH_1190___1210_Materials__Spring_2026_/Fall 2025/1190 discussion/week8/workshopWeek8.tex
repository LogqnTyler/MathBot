\begin{enumerate}

\item ({\bf\color{ForestGreen}D2}) Given $\ds f(x) =  \frac{x^2-2x+2}{x-1}. $
\enum{
\item Find $f^{\prime}(x)$ and determine all critical numbers (there are a total of $3$)
\item Construct a sign chart for $f'$ and determine which intervals the function is increasing and decreasing
\item For each critical number, decide if it is a local minimum, local maximum, or neither.
\item $\ds f^{\prime\prime}(x)=\frac{2}{(x-1)^3}$. Find the intervals where $f$ is concave up and concave down. Are there any inflection points? Why or why not?
\item Sketch the graph of $f(x)$. Note that there is a vertical asymptote at $x=1$.
}


    \vfill



\end{enumerate}

    \item Use technology to sketch a graph of $g(x)=\ln(x^2+1)-x $. What is its domain? Why is it different from the domain of $f$?
    \end{enumerate}
